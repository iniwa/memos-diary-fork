```powershell

    Set-Location -LiteralPath 'D:\Git\memos-diary-candidate-26.11.0'
    $branch = git branch --show-current
    if ($LASTEXITCODE -ne 0 -or $branch -ne 'diary-mode') { throw 'Wrong branch.' }
    $head = git rev-parse HEAD
    if ($LASTEXITCODE -ne 0 -or $head -ne 'c7aea6fe105d38aaf9c7a06ecd2500a94cd004d3') { throw 'HEAD changed.' }
    $target = git rev-parse 'refs/tags/26.11.0^{commit}'
    if ($LASTEXITCODE -ne 0 -or $target -ne 'b40d8a7b573774c52b022c8d6537d6baf2129dc0') { throw 'Run step 1 successfully first.' }
    if ((Test-Path .git/MERGE_HEAD) -or (Test-Path .git/rebase-merge) -or (Test-Path .git/rebase-apply)) { throw 'An operation is already active.' }
    git diff --cached --quiet
    if ($LASTEXITCODE -ne 0) { throw 'Index is not empty; stop.' }

    if ($LASTEXITCODE -ne 0 -or @($dirty | Where-Object { $_ -ne 'AGENTS.md' }).Count -ne 0) { throw 'Unexpected tracked changes.' }

    $saved = Join-Path (Get-Location).Path '.niwa143-host-state-26.11.0'
    if (Test-Path -LiteralPath $saved) { throw 'Recovery directory exists; do not rerun or overwrite it.' }
    New-Item -ItemType Directory -Path $saved | Out-Null
    $agentFile = Join-Path (Get-Location).Path 'AGENTS.md'
    $backup = Join-Path $saved 'AGENTS.md.raw'
    [IO.File]::WriteAllBytes($backup, [IO.File]::ReadAllBytes($agentFile))

    if ((Get-FileHash -Algorithm SHA256 -LiteralPath $backup).Hash -ne $before) { throw 'Backup hash mismatch.' }

        if ((Get-FileHash -Algorithm SHA256 -LiteralPath $agentFile).Hash -ne $before) { throw 'Instruction restore hash mismatch; retain recovery bytes.' }
    }
    Write-Output "merge exit=$mergeExit; instruction SHA256=$before"
    git status --short --branch
    git diff --name-only --diff-filter=U
    git rev-parse -q --verify MERGE_HEAD
```
